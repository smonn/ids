import { Card, CardGrid } from "@astrojs/starlight/components";

Each ID looks like `usr_01h7b3k9rqxn4cw3p9r8t2sgkw`: a three-letter brand, an
underscore, then 26 Crockford base32 characters of payload.

<CardGrid>
  <Card title="Branded & type-safe" icon="approve-check">
    `Id<"usr">` and `Id<"org">` are nominally distinct — the type system catches
    a `UserId` passed where an `OrgId` was needed.
  </Card>
  <Card title="Sortable by time" icon="seti:clock">
    The payload leads with a 48-bit millisecond timestamp, so `ORDER BY id`
    sorts by creation time. No separate `created_at` column needed.
  </Card>
  <Card title="Six codecs" icon="puzzle">
    Plaintext, newest-first, signed (tamper-evident), opaque (encrypted),
    wrapped-key, and digest (one-way keyed digest) — all sharing one wire shape.
  </Card>
  <Card title="Run it live" icon="rocket">
    The [playground](/playground/) runs the real library in your browser —
    generate, verify, unwrap, and inspect every codec.
  </Card>
</CardGrid>
